# === ImmortalWrt 官方推荐：LLVM 工具链（Ubuntu 22.04 默认装 14）===
# 用于编译内核 BPF 组件（如 bpf-headers），以及部分依赖 LLVM 的包
clang
lld
llvm

# === 额外补充：满足 bpf-headers 的 Clang/LLVM >= 15 硬性要求 ===
# 内核 6.18 起，BPF 头文件编译要求主机 Clang >= 15；
# Ubuntu 22.04 仓库默认的 clang/llvm 是 14，会在此步报
# "Minimum Clang version: 15.0.0" 并中止编译。
# 安装 15 版本后由 init.sh 通过 update-alternatives 设为默认。
clang-15
llvm-15

# === 额外补充：FUSE 用户态文件系统开发支持 ===
# 提供 fuse.h 头文件与 libfuse 链接库，供构建主机编译
# 依赖 FUSE 的软件包（如 alist、sshfs 等）使用；
# 缺少时会在编译阶段报 "fuse.h: No such file or directory"。
libfuse-dev
